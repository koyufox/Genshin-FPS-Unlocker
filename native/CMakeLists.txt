cmake_minimum_required(VERSION 4.0.2)
project(genshin_unlock LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Build with the Android NDK for arm64-v8a")
endif()
add_compile_options(-ffunction-sections -fdata-sections $<$<CONFIG:Release>:-g0>)
add_link_options(-Wl,-z,max-page-size=16384 -Wl,--gc-sections)
set(SHADOW "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/shadowhook/shadowhook/src/main/cpp")
if(NOT EXISTS "${SHADOW}/include/shadowhook.h")
    message(FATAL_ERROR "Initialize ShadowHook with: git submodule update --init --recursive")
endif()
file(GLOB SHADOW_SOURCES CONFIGURE_DEPENDS "${SHADOW}/*.c" "${SHADOW}/arch/arm64/*.c"
     "${SHADOW}/arch/arm64/*.S" "${SHADOW}/common/*.c" "${SHADOW}/third_party/*/*.c")
add_library(shadowhook SHARED ${SHADOW_SOURCES} shadowhook_loader.c)
target_include_directories(shadowhook PRIVATE "${SHADOW}" "${SHADOW}/include" "${SHADOW}/arch/arm64"
    "${SHADOW}/common" "${SHADOW}/third_party/xdl" "${SHADOW}/third_party/bsd" "${SHADOW}/third_party/lss")
target_compile_features(shadowhook PRIVATE c_std_11)
set_source_files_properties(shadowhook_loader.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
# Keep modern-NDK warning policy separate from the module's warnings-as-errors.
target_compile_options(shadowhook PRIVATE -Os)
target_compile_definitions(shadowhook PRIVATE $<$<CONFIG:Debug>:SH_CONFIG_DEBUG>)
target_link_libraries(shadowhook PRIVATE log dl)
file(READ "${SHADOW}/shadowhook.map.txt" SHADOW_EXPORTS)
string(REPLACE "        shadowhook_init;" "        shadowhook_init;\n        genshin_shadowhook_init;" SHADOW_EXPORTS "${SHADOW_EXPORTS}")
string(REPLACE "        shadowhook_interceptor_glue_vfpv3d16;" "" SHADOW_EXPORTS "${SHADOW_EXPORTS}")
string(REPLACE "        shadowhook_interceptor_glue_vfpv3d32;" "" SHADOW_EXPORTS "${SHADOW_EXPORTS}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/shadowhook-arm64.map" "${SHADOW_EXPORTS}")
target_link_options(shadowhook PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_BINARY_DIR}/shadowhook-arm64.map" -Wl,--exclude-libs,ALL
    -Wl,--wrap=dlopen $<$<CONFIG:Release>:-Wl,--wrap=__android_log_print>)
add_library(shadowhook_nothing SHARED "${SHADOW}/nothing/sh_nothing.c")
target_compile_options(shadowhook_nothing PRIVATE -Oz -fno-ident -fno-unwind-tables -fno-asynchronous-unwind-tables)
target_link_options(shadowhook_nothing PRIVATE -Wl,--as-needed -Wl,--hash-style=sysv -Wl,--build-id=none
    -Wl,-nostdlib -nostartfiles -nodefaultlibs)

if(NOT EXISTS "${RUST_LIBRARY}")
    message(FATAL_ERROR "Build the Rust static library through build.py first")
endif()
add_library(unlock_rust STATIC IMPORTED)
set_target_properties(unlock_rust PROPERTIES IMPORTED_LOCATION "${RUST_LIBRARY}")
add_library(genshin_unlock SHARED module.cpp)
target_compile_options(genshin_unlock PRIVATE -Wall -Wextra -Werror -fno-exceptions -fno-rtti -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_libraries(genshin_unlock PRIVATE unlock_rust log dl m)
target_link_options(genshin_unlock PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/module.map" -Wl,--exclude-libs,ALL)

add_executable(genshin_control control.c)
target_compile_options(genshin_control PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(genshin_control PRIVATE unlock_rust log dl m)
target_link_options(genshin_control PRIVATE -Wl,--exclude-libs,ALL)
